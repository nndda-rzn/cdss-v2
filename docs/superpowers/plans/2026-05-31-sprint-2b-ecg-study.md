    encounter_id: Mapped[int] = mapped_column(

"""Path construction, study UID generation, and disk persistence for ECG images."""
import secrets
from datetime import UTC, datetime
from pathlib import Path

_EXT_BY_FORMAT = {"jpeg": "jpg", "png": "png"}


def generate_study_uid(facility_code: str) -> str:
    ts = datetime.now(UTC).strftime("%Y%m%d%H%M%S")
    rand = secrets.token_hex(2)  # 4 hex chars
    return f"ECG-{facility_code}-{ts}-{rand}"


def storage_path_for(
    *, facility_id: int, encounter_id: int, study_uid: str, ext: str
) -> str:
    """Relative path under the storage root."""
    return f"uploads/ecg/{facility_id}/{encounter_id}/{study_uid}.{ext}"


def save_image(storage_root: Path, relative_path: str, data: bytes) -> Path:
    """Write bytes to storage_root/relative_path, creating parent dirs."""
    target = storage_root / relative_path
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(data)
    return target
```

"""Validate an uploaded image by size and magic bytes (not the client MIME)."""
from app.core.exceptions.base import ValidationError

_MAX_BYTES = 10 * 1024 * 1024
_MIN_BYTES = 1024

_MAGIC = {
    "jpeg": (b"\xff\xd8\xff",),
    "png": (b"\x89PNG\r\n\x1a\n",),
}
_EXT = {"jpeg": "jpg", "png": "png"}


def validate_image(data: bytes, *, content_type: str | None = None) -> tuple[str, str]:
    """Return (format, ext) for a valid JPEG/PNG; raise ValidationError otherwise.

    The client-supplied content_type is advisory only; the format is decided by
    magic bytes to prevent spoofing.
    """
    size = len(data)
    if size < _MIN_BYTES:
        raise ValidationError(
            code="FILE_TOO_SMALL",
            message_id="Berkas terlalu kecil",
            message_en="File is too small",
        )
    if size > _MAX_BYTES:
        raise ValidationError(
            code="FILE_TOO_LARGE",
            message_id="Berkas melebihi 10 MB",
            message_en="File exceeds 10 MB",
        )
    for fmt, signatures in _MAGIC.items():
        if any(data.startswith(sig) for sig in signatures):
            return fmt, _EXT[fmt]
    raise ValidationError(
        code="INVALID_FILE_FORMAT",
        message_id="Format berkas harus JPEG atau PNG",
        message_en="File format must be JPEG or PNG",
    )
```

```python
"""Compute an image-quality score and status from blur/brightness/contrast/noise."""
from typing import overload

import cv2
import numpy as np

_WEIGHTS = {"blur": 0.35, "contrast": 0.25, "brightness": 0.20, "noise": 0.20}


def _decode_gray(data: bytes) -> np.ndarray:
    arr = np.frombuffer(data, dtype=np.uint8)
    img = cv2.imdecode(arr, cv2.IMREAD_GRAYSCALE)
    if img is None:
        raise ValueError("Unable to decode image for quality assessment")
    return img


def _metrics(gray: np.ndarray) -> dict[str, float]:
    blur = float(cv2.Laplacian(gray, cv2.CV_64F).var())
    brightness = float(gray.mean())
    contrast = float(gray.std())
    smoothed = cv2.GaussianBlur(gray, (3, 3), 0)
    noise = float(np.abs(gray.astype(np.float64) - smoothed.astype(np.float64)).std())
    return {"blur": blur, "noise": noise,
            "brightness": brightness, "contrast": contrast}


def _score(metrics: dict[str, float]) -> tuple[float, str]:
    sub = {
        "blur": min(metrics["blur"] / 100.0, 1.0),
        "brightness": max(1.0 - abs(metrics["brightness"] - 130.0) / 130.0, 0.0),
        "contrast": min(metrics["contrast"] / 60.0, 1.0),
        "noise": max(1.0 - metrics["noise"] / 30.0, 0.0),
    }
    composite = sum(sub[k] * _WEIGHTS[k] for k in _WEIGHTS)
    composite = round(composite, 3)
    status = "good" if composite >= 0.7 else "acceptable" if composite >= 0.4 else "poor"
    return composite, status


@overload
def assess_quality(data: bytes) -> tuple[float, str]: ...
@overload
def assess_quality(
    data: bytes, *, detailed: bool
) -> tuple[float, str, dict[str, float]]: ...


def assess_quality(
    data: bytes, *, detailed: bool = False
) -> tuple[float, str] | tuple[float, str, dict[str, float]]:
    gray = _decode_gray(data)
    metrics = _metrics(gray)
    score, status = _score(metrics)
    if detailed:
        return score, status, metrics
    return score, status
```

"""Ensure ai_analysis consent before an ECG upload (progressive capture)."""
import uuid

from sqlalchemy import select
from sqlalchemy.orm import Session

from app.core.exceptions.base import ValidationError
from app.db.models.consent import Consent
from app.services.consent.capture import capture_consent


def ensure_ai_analysis_consent(
    session: Session, *, patient_id: uuid.UUID, user_id: int
) -> None:
    """Auto-capture ai_analysis consent if none exists; honor explicit revoke."""
    latest = session.scalar(
        select(Consent)
        .where(
            Consent.patient_id == patient_id,
            Consent.consent_type == "ai_analysis",
        )
        .order_by(Consent.id.desc())
        .limit(1)
    )
    if latest is None:
        capture_consent(
            session, patient_id=patient_id,
            consent_type="ai_analysis", user_id=user_id,
        )
        return
    if latest.status == "revoked":
        raise ValidationError(
            code="CONSENT_REVOKED",
            message_id="Persetujuan analisis AI telah dicabut",
            message_en="ai_analysis consent has been revoked",
        )
    # status == "given" (or "pending"): no-op for "given"; treat pending as block-free
    # capture if pending so the row reflects an explicit grant.
    if latest.status == "pending":
        capture_consent(
            session, patient_id=patient_id,
            consent_type="ai_analysis", user_id=user_id,
        )
```

import { useAuthStore } from "@/features/auth/store";
import { apiFetch } from "@/lib/api/client";

export interface EcgStudyRow {
  id: number;
  encounter_id: number;
  patient_id?: string;
  study_uid: string;
  image_original_path: string;
  image_processed_path: string | null;
  image_quality_status: "good" | "acceptable" | "poor";
  image_quality_score: number;
  image_quality_metrics_json: Record<string, number>;
  acquired_at: string;
  recorded_by_user_id: number;
  device_info?: string | null;
  created_at: string;
}

export const ecgKeys = {
  list: (encounterId: string) => ["ecg-studies", encounterId] as const,
};

export async function fetchEcgStudies(
  encounterId: string,
): Promise<EcgStudyRow[]> {
  const resp = await apiFetch(`/api/encounters/${encounterId}/ecg-studies`);
  const body = await resp.json();
  if (!resp.ok) throw new Error(body.error?.message_id ?? "Gagal memuat studi EKG");
  return body.items as EcgStudyRow[];
}

export async function uploadEcgStudy(
  encounterId: string,
  file: File,
  deviceInfo?: string,
): Promise<void> {
  const form = new FormData();
  form.append("file", file);
  if (deviceInfo) form.append("device_info", deviceInfo);

  const { accessToken } = useAuthStore.getState();
  const headers: Record<string, string> = {};
  if (accessToken) headers["Authorization"] = `Bearer ${accessToken}`;

  const resp = await fetch(`/api/encounters/${encounterId}/ecg-studies`, {
    method: "POST",
    headers,
    body: form,
  });
  if (!resp.ok) {
    const body = await resp.json().catch(() => ({}));
    throw new Error(body.error?.message_id ?? "Gagal mengunggah EKG");
  }
}

    queryFn: () => fetchEcgStudies(encounterId),
  });

  if (!data || data.length === 0) {
    return <p className="text-sm text-muted-foreground">Belum ada studi EKG.</p>;
  }

  return (
    <table className="w-full text-sm">
      <thead>
        <tr className="border-b text-left">
          <th className="py-1">Study UID</th>
          <th className="py-1">Kualitas</th>
          <th className="py-1">Skor</th>
          <th className="py-1">Waktu</th>
        </tr>
      </thead>
      <tbody>
        {data.map((s) => (
          <tr key={s.id} className="border-b">
            <td className="py-1 font-mono text-xs">{s.study_uid}</td>
            <td className="py-1">
              <Badge variant={_STATUS_VARIANT[s.image_quality_status]}>
                {_STATUS_LABEL[s.image_quality_status]}
              </Badge>
            </td>
            <td className="py-1">{s.image_quality_score.toFixed(2)}</td>
            <td className="py-1">{s.created_at.slice(0, 16)}</td>
          </tr>
        ))}
      </tbody>
    </table>
  );
}
```

Create `frontend/src/features/ecg_study/EcgUploadForm.tsx`:

```tsx
import { useRef, useState } from "react";
import { useMutation, useQueryClient } from "@tanstack/react-query";
import { Button } from "@/components/ui/button";
import { ecgKeys, uploadEcgStudy } from "./api";

const _ALLOWED = ["image/jpeg", "image/png"];

export function EcgUploadForm({
  encounterId,
  onUploadSuccess,
}: {
  encounterId: string;
  onUploadSuccess?: () => void;
}) {
  const qc = useQueryClient();
  const inputRef = useRef<HTMLInputElement>(null);
  const [file, setFile] = useState<File | null>(null);
  const [preview, setPreview] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);

  const mutation = useMutation({
    mutationFn: () => {
      if (!file) throw new Error("Tidak ada berkas");
      return uploadEcgStudy(encounterId, file);
    },
    onSuccess: () => {
      setFile(null);
      setPreview(null);
      if (inputRef.current) inputRef.current.value = "";
      qc.invalidateQueries({ queryKey: ecgKeys.list(encounterId) });
      onUploadSuccess?.();
    },
    onError: (err) =>
      setError(err instanceof Error ? err.message : "Gagal mengunggah"),
  });

  function onSelect(e: React.ChangeEvent<HTMLInputElement>) {
    setError(null);
    const f = e.target.files?.[0] ?? null;
    if (!f) return;
    if (!_ALLOWED.includes(f.type)) {
      setError("Format harus JPEG atau PNG");
      setFile(null);
      setPreview(null);
      return;
    }
    if (f.size > 10 * 1024 * 1024) {
      setError("Berkas melebihi 10 MB");
      setFile(null);
      setPreview(null);
      return;
    }
    setFile(f);
    setPreview(URL.createObjectURL(f));
  }

  return (
    <div className="space-y-3">
      <label className="text-sm font-medium" htmlFor="ecg-file">
        Pilih berkas EKG (JPEG/PNG, maks 10 MB)
      </label>
      <input
        id="ecg-file"
        ref={inputRef}
        type="file"
        accept="image/jpeg,image/png"
        onChange={onSelect}
        className="block text-sm"
      />
      {preview && (
        <img
          src={preview}
          alt="Pratinjau EKG"
          className="max-h-48 rounded border"
        />
      )}
      {error && (
        <p role="alert" className="text-sm text-destructive">
          {error}
        </p>
      )}
      <Button
        disabled={!file || mutation.isPending}
        onClick={() => mutation.mutate()}
      >
        {mutation.isPending ? "Mengunggah..." : "Unggah EKG"}
      </Button>
    </div>
  );
}
```
